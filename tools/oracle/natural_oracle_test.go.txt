func localeInputs(d *locale.Data) []string {
	var out []string
	r := d.Relative
	first := func(xs []string) string {
		if len(xs) == 0 {
			return ""
		}
		return xs[0]
	}
	add := func(s string) {
		if s != "" {
			out = append(out, s, strings.ToUpper(s))
		}
	}
	for _, list := range [][]string{r.Now, r.Today, r.Yesterday, r.Tomorrow} {
		for _, w := range list {
			add(w)
			add(w + " 5pm")
			add(w + " at 14:00")
		}
	}
	units := [][]string{r.Seconds, r.Minutes, r.Hours, r.Days, r.Weeks, r.Months, r.Years}
	for _, list := range units {
		for _, u := range list {
			for _, ago := range r.Ago {
				add("3 " + u + " " + ago)
				add(ago + " 3 " + u)
			}
			for _, in := range r.InFuture {
				add(in + " 3 " + u)
			}
			for _, sel := range [][]string{r.Last, r.Next, r.This} {
				add(first(sel) + " " + u)
			}
		}
	}
	for i := range 7 {
		for _, wd := range []string{d.WeekdaysWide[i], d.WeekdaysAbbr[i]} {
			add(wd)
			for _, sel := range [][]string{r.Last, r.Next, r.This} {
				for _, s := range sel {
					add(s + " " + wd)
				}
			}
		}
	}
	return out
}

func formatResult(r *Result) string {
	if r == nil {
		return "nil"
	}
	f := func(t time.Time) string {
		y, m, d := t.Date()
		hh, mm, ss := t.Clock()
		_, off := t.Zone()
		return fmt.Sprintf("%d-%02d-%02dT%02d:%02d:%02d.%09d%+d", y, int(m), d, hh, mm, ss, t.Nanosecond(), off)
	}
	alt := "-"
	if r.Ambiguous {
		alt = f(r.AltTime)
	}
	return fmt.Sprintf("%s kind=%d amb=%t alt=%s", f(r.Time), int(r.Kind), r.Ambiguous, alt)
}

func TestZZOracle(t *testing.T) {
	var all []*locale.Data
	for _, tag := range locale.Tags() {
		all = append(all, locale.Lookup(tag))
	}
	w := bufio.NewWriter(os.Stdout)
	path := os.Getenv("ORACLE_OUT")
	f, err := os.Create(path)
	if err != nil {
		t.Fatal(err)
	}
	defer f.Close()
	w = bufio.NewWriter(f)
	defer w.Flush()
	emit := func(spec string, locs []*locale.Data, in string) {
		for bi, base := range oracleBases {
			for _, pf := range []bool{false, true} {
				r := Parse(in, Config{BaseTime: base, PreferFuture: pf, Locales: locs})
				p := 0
				if pf {
					p = 1
				}
				fmt.Fprintf(w, "%d %d %s %s %s\n", bi, p, spec, hex.EncodeToString([]byte(in)), formatResult(r))
			}
		}
	}
	for _, in := range english {
		emit("-", nil, in)
		emit("*", all, in)
	}
	if extra := os.Getenv("ORACLE_EXTRA"); extra != "" {
		data, err := os.ReadFile(extra)
		if err != nil {
			t.Fatal(err)
		}
		for _, line := range strings.Split(strings.TrimSpace(string(data)), "\n") {
			spec, in, _ := strings.Cut(line, "\t")
			if spec == "-" {
				emit("-", nil, in)
			} else {
				emit(spec, []*locale.Data{locale.Lookup(spec)}, in)
			}
		}
	}
	for _, d := range all {
		for _, in := range localeInputs(d) {
			emit(d.Tag, []*locale.Data{d}, in)
			emit("*", all, in)
		}
	}
}
